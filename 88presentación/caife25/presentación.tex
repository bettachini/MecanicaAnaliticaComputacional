\begin{frame}
	\frametitle{El código como competencia del mecánico}
	\begin{block}{Enfoque construccionista}
		Papert (1980) ``El mejor aprendizaje tiene lugar cuando el sujeto toma el mando''
	\end{block}
	\pause
	\begin{block}{El alumno ``construye" su competencia}
		\begin{itemize}[<+->]
			\item Problemas de ejemplo se resuelven con el código provisto por el profesor
			\item El estudiante lo \textbf{recicla} para resolver los de la guía de ejercicios
			\item Gradualmente adquiere autonomía y genera su propio código.
			\item Finalmente, puede aprovecharse de la IA para acelerar su trabajo.
			\item Tras cursar conserva su repositorio con todo su código.
		\end{itemize}
	\end{block}
\end{frame}
